### 3. 自动化测试

部署前必须有测试。新建 `test_app.py`:

python

```python
import pytest
from fastapi.testclient import TestClient

from app import app


@pytest.fixture(scope="module")
def client():
    with TestClient(app) as c:        # with 会触发 lifespan,把模型真正加载进来
        yield c


def test_health(client):
    r = client.get("/health")
    assert r.status_code == 200
    assert r.json()["status"] == "ok"


def test_predict_format(client):
    r = client.post("/predict", json={"texts": ["很好", "很差"]})
    assert r.status_code == 200
    preds = r.json()["predictions"]
    assert len(preds) == 2
    for p in preds:
        assert p["label"] in ("正面", "负面")
        assert 0.0 <= p["confidence"] <= 1.0


def test_obvious_cases(client):
    """冒烟测试:几个毫无歧义的例子必须对。这类测试能拦住'模型文件被换错'这种低级事故"""
    r = client.post("/predict", json={"texts": [
        "非常满意,环境优美,服务周到,强烈推荐",
        "糟糕透顶,又脏又吵,服务员态度恶劣,绝不再来",
    ]})
    labels = [p["label"] for p in r.json()["predictions"]]
    assert labels == ["正面", "负面"]


@pytest.mark.parametrize("payload", [
    {"texts": []},                    # 空列表
    {"texts": ["   "]},               # 全是空白
    {"texts": ["a"] * 100},           # 超过批量上限
    {"texts": ["字" * 5000]},          # 超长
    {},                               # 缺少字段
    {"texts": "不是列表"},             # 类型错误
])
def test_bad_input_rejected(client, payload):
    assert client.post("/predict", json=payload).status_code == 422


def test_batch_matches_single(client):
    """同一条文本,单条预测和放在批次里预测,结果应一致(检查 padding 没有引入 bug)"""
    text = "位置方便,但是早餐很一般"
    single = client.post("/predict", json={"texts": [text]}).json()["predictions"][0]
    batch = client.post("/predict", json={"texts": [text, "完全不相关的另一句很长很长的话" * 3]}).json()["predictions"][0]
    assert single["label_id"] == batch["label_id"]
    assert abs(single["confidence"] - batch["confidence"]) < 1e-2
```

**机器学习服务的测试和普通软件有区别:** 你无法断言"输出恰好等于某个值",所以测的是**格式正确、边界输入被拒绝、明显的例子必须对、不同调用方式结果一致**。
